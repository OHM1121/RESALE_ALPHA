    def on_upbit(self,item):

        with self.cv:
            if typ=='myOrder':
                oid=str(item.get('uuid') or '');ident=str(item.get('identifier') or '')

                if ident:self.coin_ids[ident]=dict(item)

class UpbitBroker:
    REST='https://api.upbit.com'
    def __init__(self,access,secret,journal,tracker=None):
        self.access=access or '';self.secret=secret or '';self.journal=journal;self.tracker=tracker
        import requests
        self.s=requests.Session()
    def ready(self):return bool(self.access and self.secret)
    @staticmethod
    def _b64(raw):
        import base64
        return base64.urlsafe_b64encode(raw).decode().rstrip('=')
    def _jwt(self,params=None):
        head=self._b64(b'{"alg":"HS256","typ":"JWT"}')
        payload={'access_key':self.access,'nonce':str(uuid.uuid4())}
        if params:
            q=urlencode([(k,str(v)) for k,v in params.items()])
            payload['query_hash']=hashlib.sha512(q.encode()).hexdigest();payload['query_hash_alg']='SHA512'
        body=self._b64(json.dumps(payload,separators=(',',':')).encode())
        sig=hmac.new(self.secret.encode(),(head+'.'+body).encode(),hashlib.sha256).digest()
        return head+'.'+body+'.'+self._b64(sig)
