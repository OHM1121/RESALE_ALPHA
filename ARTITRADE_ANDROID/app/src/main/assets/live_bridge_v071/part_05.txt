    def _upbit_private_loop(self):
        if not self.upbit.ready():return
        try:from websockets.sync.client import connect
        except Exception as e:self.feed_tracker.coin_connected_set(False,'IMPORT:'+type(e).__name__);self._err('upbit_private_import',e);return
        backoff=2
        while not self._private_stop.is_set():
            try:
                token=self.upbit._jwt();headers={'Authorization':'Bearer '+token}
                kwargs=dict(open_timeout=10,ping_interval=20,ping_timeout=15,max_size=2**20,close_timeout=5)
                try:wsctx=connect('wss://api.upbit.com/websocket/v1/private',additional_headers=headers,**kwargs)
                except TypeError:wsctx=connect('wss://api.upbit.com/websocket/v1/private',extra_headers=headers,**kwargs)
                with wsctx as ws:
                    sub=[{'ticket':'ARTITRADE_PRIVATE_'+uuid.uuid4().hex},{'type':'myOrder'},{'type':'myAsset'},{'format':'DEFAULT'}]
                    ws.send(json.dumps(sub,separators=(',',':')));self.feed_tracker.coin_connected_set(True);backoff=2
                    while not self._private_stop.is_set():
                        try:raw=ws.recv(timeout=2)
                        except TimeoutError:continue
                        if raw is None:raise RuntimeError('PRIVATE_WS_EOF')
                        item=json.loads(raw.decode('utf-8') if isinstance(raw,bytes) else raw)
                        if isinstance(item,dict) and item.get('error'):raise RuntimeError('UPBIT_PRIVATE:'+str(item.get('error'))[:180])
                        self.feed_tracker.on_upbit(item)
            except Exception as e:
                if self._private_stop.is_set():break
                self.feed_tracker.coin_connected_set(False,type(e).__name__+':'+str(e)[:160]);self._err('upbit_private_ws',e)
                self._private_stop.wait(backoff);backoff=min(30,backoff*2)
        self.feed_tracker.coin_connected_set(False,'STOPPED')
